"""
Tabular Q-learning agent.

This is the base reinforcement-learning algorithm.

The agent uses:

    Q(s, a)

with:

    epsilon-greedy exploration
    learning rate
    discount factor
    temporal-difference update
"""

from __future__ import annotations


import pickle
import random
from collections import defaultdict

import numpy as np


class QLearningAgent:

    def __init__(
        self,
        action_count: int = 5,
        learning_rate: float = 0.1,
        discount_factor: float = 0.95,
        epsilon: float = 1.0,
        epsilon_min: float = 0.05,
        epsilon_decay: float = 0.995,
        seed: int = 42,
        
    ):
        self.state_decimals = 2
        
        if action_count <= 0:
            raise ValueError(
                "action_count must be positive."
            )

        self.action_count = int(
            action_count
        )

        self.learning_rate = float(
            learning_rate
        )

        self.discount_factor = float(
            discount_factor
        )

        self.epsilon = float(
            epsilon
        )

        self.epsilon_min = float(
            epsilon_min
        )

        self.epsilon_decay = float(
            epsilon_decay
        )

        self.random = random.Random(
            seed
        )

        # --------------------------------------------------
        # Q-table
        # --------------------------------------------------

        self.q_table = defaultdict(
            self._new_q_values
        )

        self.training_steps = 0

    # ==================================================
    # DEFAULT Q VALUES
    # ==================================================

    def _new_q_values(self):

        return np.zeros(
            self.action_count,
            dtype=np.float64,
        )

    # ==================================================
    # STATE KEY
    # ==================================================

    def state_key(
        self,
        state,
    ):

        if isinstance(
            state,
            (np.ndarray, list, tuple),
        ):

            values = np.asarray(
                state,
                dtype=np.float64,
            )

            return tuple(
                np.round(
                    values,
                    self.state_decimals,
                ).tolist()
            )

        return state

    # ==================================================
    # ACTION SELECTION
    # ==================================================

    def select_action(
        self,
        state,
        training=True,
    ):

        key = self.state_key(
            state
        )

        # --------------------------------------------------
        # Exploration
        # --------------------------------------------------

        if (
            training
            and
            self.random.random()
            <
            self.epsilon
        ):

            return self.random.randrange(
                self.action_count
            )

        # --------------------------------------------------
        # Exploitation
        # --------------------------------------------------

        q_values = self.q_table[
            key
        ]

        max_value = np.max(
            q_values
        )

        best_actions = np.flatnonzero(
            np.isclose(
                q_values,
                max_value,
            )
        )

        return int(
            self.random.choice(
                best_actions.tolist()
            )
        )

    # ==================================================
    # Q UPDATE
    # ==================================================

    def update(
        self,
        state,
        action,
        reward,
        next_state,
        terminated,
    ):

        state_key = self.state_key(
            state
        )

        next_state_key = self.state_key(
            next_state
        )

        action = int(
            action
        )

        if not (
            0
            <=
            action
            <
            self.action_count
        ):

            raise ValueError(
                f"Invalid action {action}."
            )

        current_q = self.q_table[
            state_key
        ][
            action
        ]

        # --------------------------------------------------
        # Bellman target
        # --------------------------------------------------

        if terminated:

            target = float(
                reward
            )

        else:

            next_max = np.max(
                self.q_table[
                    next_state_key
                ]
            )

            target = (
                float(reward)
                +
                self.discount_factor
                *
                next_max
            )

        # --------------------------------------------------
        # Temporal-difference update
        # --------------------------------------------------

        td_error = (
            target
            -
            current_q
        )

        self.q_table[
            state_key
        ][
            action
        ] += (
            self.learning_rate
            *
            td_error
        )

        self.training_steps += 1

        return float(
            td_error
        )

    # ==================================================
    # EPSILON DECAY
    # ==================================================

    def decay_epsilon(self):

        self.epsilon = max(
            self.epsilon_min,
            self.epsilon
            *
            self.epsilon_decay,
        )

        return self.epsilon

    # ==================================================
    # GREEDY ACTION
    # ==================================================

    def greedy_action(
        self,
        state,
    ):

        return self.select_action(
            state,
            training=False,
        )

    # ==================================================
    # TABLE SIZE
    # ==================================================

    @property
    def state_count(self):

        return len(
            self.q_table
        )

    # ==================================================
    # RESET
    # ==================================================

    def clear(self):

        self.q_table.clear()

        self.training_steps = 0

        # ==================================================
    # SAVE
    # ==================================================

    def save(
        self,
        filepath,
    ):

        data = {
            "q_table": dict(
                self.q_table
            ),
            "action_count":
                self.action_count,
            "learning_rate":
                self.learning_rate,
            "discount_factor":
                self.discount_factor,
            "epsilon":
                self.epsilon,
            "epsilon_min":
                self.epsilon_min,
            "epsilon_decay":
                self.epsilon_decay,
            "training_steps":
                self.training_steps,
        }

        with open(
            filepath,
            "wb",
        ) as file:

"""
Tabular Q-learning agent.

This is the base reinforcement-learning algorithm.

The agent uses:

    Q(s, a)

with:

    epsilon-greedy exploration
    learning rate
    discount factor
    temporal-difference update
"""

from __future__ import annotations


import pickle
import random
from collections import defaultdict

import numpy as np


class QLearningAgent:

    def __init__(
        self,
        action_count: int = 5,
        learning_rate: float = 0.1,
        discount_factor: float = 0.95,
        epsilon: float = 1.0,
        epsilon_min: float = 0.05,
        epsilon_decay: float = 0.995,
        seed: int = 42,
    ):

        if action_count <= 0:
            raise ValueError(
                "action_count must be positive."
            )

        self.action_count = int(
            action_count
        )

        self.learning_rate = float(
            learning_rate
        )

        self.discount_factor = float(
            discount_factor
        )

        self.epsilon = float(
            epsilon
        )

        self.epsilon_min = float(
            epsilon_min
        )

        self.epsilon_decay = float(
            epsilon_decay
        )

        self.random = random.Random(
            seed
        )

        # --------------------------------------------------
        # Q-table
        # --------------------------------------------------

        self.q_table = defaultdict(
            self._new_q_values
        )

        self.training_steps = 0

    # ==================================================
    # DEFAULT Q VALUES
    # ==================================================

    def _new_q_values(self):

        return np.zeros(
            self.action_count,
            dtype=np.float64,
        )

    # ==================================================
    # STATE KEY
    # ==================================================

    def state_key(
        self,
        state,
    ):

        if isinstance(
            state,
            np.ndarray,
        ):

            return tuple(
                np.round(
                    state.astype(
                        np.float64
                    ),
                    6,
                ).tolist()
            )

        if isinstance(
            state,
            (list, tuple),
        ):

            return tuple(
                state
            )

        return state

    # ==================================================
    # ACTION SELECTION
    # ==================================================

    def select_action(
        self,
        state,
        training=True,
    ):

        key = self.state_key(
            state
        )

        # --------------------------------------------------
        # Exploration
        # --------------------------------------------------

        if (
            training
            and
            self.random.random()
            <
            self.epsilon
        ):

            return self.random.randrange(
                self.action_count
            )

        # --------------------------------------------------
        # Exploitation
        # --------------------------------------------------

        q_values = self.q_table[
            key
        ]

        max_value = np.max(
            q_values
        )

        best_actions = np.flatnonzero(
            np.isclose(
                q_values,
                max_value,
            )
        )

        return int(
            self.random.choice(
                best_actions.tolist()
            )
        )

    # ==================================================
    # Q UPDATE
    # ==================================================

    def update(
        self,
        state,
        action,
        reward,
        next_state,
        terminated,
    ):

        state_key = self.state_key(
            state
        )

        next_state_key = self.state_key(
            next_state
        )

        action = int(
            action
        )

        if not (
            0
            <=
            action
            <
            self.action_count
        ):

            raise ValueError(
                f"Invalid action {action}."
            )

        current_q = self.q_table[
            state_key
        ][
            action
        ]

        # --------------------------------------------------
        # Bellman target
        # --------------------------------------------------

        if terminated:

            target = float(
                reward
            )

        else:

            next_max = np.max(
                self.q_table[
                    next_state_key
                ]
            )

            target = (
                float(reward)
                +
                self.discount_factor
                *
                next_max
            )

        # --------------------------------------------------
        # Temporal-difference update
        # --------------------------------------------------

        td_error = (
            target
            -
            current_q
        )

        self.q_table[
            state_key
        ][
            action
        ] += (
            self.learning_rate
            *
            td_error
        )

        self.training_steps += 1

        return float(
            td_error
        )

    # ==================================================
    # EPSILON DECAY
    # ==================================================

    def decay_epsilon(self):

        self.epsilon = max(
            self.epsilon_min,
            self.epsilon
            *
            self.epsilon_decay,
        )

        return self.epsilon

    # ==================================================
    # GREEDY ACTION
    # ==================================================

    def greedy_action(
        self,
        state,
    ):

        return self.select_action(
            state,
            training=False,
        )

    # ==================================================
    # TABLE SIZE
    # ==================================================

    @property
    def state_count(self):

        return len(
            self.q_table
        )

    # ==================================================
    # RESET
    # ==================================================

    def clear(self):

        self.q_table.clear()

        self.training_steps = 0

        # ==================================================
    # SAVE
    # ==================================================

    def save(
        self,
        filepath,
    ):

        data = {
            "q_table": dict(
                self.q_table
            ),
            "action_count":
                self.action_count,
            "learning_rate":
                self.learning_rate,
            "discount_factor":
                self.discount_factor,
            "epsilon":
                self.epsilon,
            "epsilon_min":
                self.epsilon_min,
            "epsilon_decay":
                self.epsilon_decay,
            "training_steps":
                self.training_steps,
        }

        with open(
            filepath,
            "wb",
        ) as file:

            pickle.dump(
                data,
                file,
            )

    # ==================================================
    # LOAD
    # ==================================================

    def load(
        self,

    ):

        with open(

            "rb",
        ) as file:

            data = pickle.load(

"""

"""

from __future__ import annotations

import numpy as np

from algorithms.q_learning import QLearningAgent
from environment.multi_agv_env import MultiAGVEnv


class QLearningTrainer:

    def __init__(
        self,
        episodes: int = 100,
        max_steps: int = 500,
        num_agvs: int = 3,
        seed: int = 42,
    ):

        self.episodes = int(
            episodes
        )

        self.max_steps = int(
            max_steps
        )

        self.num_agvs = int(
            num_agvs
        )

        self.seed = int(
            seed
        )

        # --------------------------------------------------
        # Environment
        # --------------------------------------------------

        self.env = MultiAGVEnv(
            width=20,
            height=20,
            num_agvs=self.num_agvs,
            max_steps=self.max_steps,
            risk_aware_assignment=True,
        )

        # --------------------------------------------------
        # One Q-learning agent per AGV
        # --------------------------------------------------

        self.agents = [
            QLearningAgent(
                action_count=5,
                learning_rate=0.1,
                discount_factor=0.95,
                epsilon=1.0,
                epsilon_min=0.05,
                epsilon_decay=0.995,
                seed=self.seed + agv_id,
            )
            for agv_id in range(
                self.num_agvs
            )
        ]

        self.episode_rewards = []

        self.episode_lengths = []

        self.episode_completions = []

    # ==================================================
    # TRAIN ONE EPISODE
    # ==================================================

    def train_episode(
        self,
        episode: int,
    ):

        states, _ = self.env.reset(
            seed=self.seed + episode
        )

        total_reward = 0.0

        completed_tasks = 0

        for step in range(
            self.max_steps
        ):

            # --------------------------------------------------
            # Select action independently for every AGV
            # --------------------------------------------------

            actions = np.array(
                [
                    agent.select_action(
                        self._get_agv_state(
                            states,
                            agv_id,
                        ),
                        training=True,
                    )
                    for agv_id, agent
                    in enumerate(
                        self.agents
                    )
                ],
                dtype=np.int64,
            )

            (
                next_states,
                reward,
                terminated,
                truncated,
                info,
            ) = self.env.step(
                actions
            )

            # --------------------------------------------------
            # Update each Q learner
            # --------------------------------------------------

            for agv_id, agent in enumerate(
                self.agents
            ):

                state = self._get_agv_state(
                    states,
                    agv_id,
                )

                next_state = self._get_agv_state(
                    next_states,
                    agv_id,
                )

                agent.update(
                    state=state,
                    action=int(
                        actions[agv_id]
                    ),
                    reward=reward,
                    next_state=next_state,
                    terminated=(
                        terminated
                        or truncated
                    ),
                )

            states = next_states

            total_reward += float(
                reward
            )

            completed_tasks = int(
                info.get(
                    "completed_tasks",
                    0,
                )
            )

            if terminated or truncated:

                break

        # --------------------------------------------------
        # Exploration decay
        # --------------------------------------------------

        for agent in self.agents:

            agent.decay_epsilon()

        self.episode_rewards.append(
            total_reward
        )

        self.episode_lengths.append(
            step + 1
        )

        self.episode_completions.append(
            completed_tasks
        )

        return {
            "reward":
                total_reward,

            "steps":
                step + 1,

            "completed":
                completed_tasks,

            "epsilon":
                self.agents[0].epsilon,
        }

    # ==================================================
    # EVALUATE
    # ==================================================
    def evaluate(
        self,
        episodes=20,
        max_steps=None,
        seed=None,
    ):
        max_steps = int(max_steps or self.max_steps)
        base_seed = self.seed if seed is None else int(seed)
        rewards = []
        lengths = []
        completions = []
        collisions = []

        for episode in range(int(episodes)):
            states, _ = self.env.reset(seed=base_seed + episode)
            total_reward = 0.0
            completed = 0
            collision_count = 0

            for step in range(max_steps):
                actions = np.array([
                    agent.select_action(
                        self._get_agv_state(states, agv_id),
                        training=False,
                    )
                    for agv_id, agent in enumerate(self.agents)
                ], dtype=np.int64)

                next_states, reward, terminated, truncated, info = self.env.step(actions)
                total_reward += float(reward)
                completed = int(info.get('completed_tasks', completed))
                collision_count = int(info.get('episode_collision_count', info.get('collision_count', collision_count)))
                states = next_states

                if terminated or truncated:
                    break

            rewards.append(total_reward)
            lengths.append(step + 1)
            completions.append(completed)
            collisions.append(collision_count)

        return {
            'episodes': int(episodes),
            'max_steps': max_steps,
            'average_reward': float(np.mean(rewards)) if rewards else 0.0,
            'average_steps': float(np.mean(lengths)) if lengths else 0.0,
            'average_completed': float(np.mean(completions)) if completions else 0.0,
            'completion_rate': float(100.0 * np.mean(np.asarray(completions) > 0)) if completions else 0.0,
            'average_collisions': float(np.mean(collisions)) if collisions else 0.0,
            'rewards': rewards,
            'steps': lengths,
            'completions': completions,
            'collisions': collisions,
        }

    # ==================================================
    # TRAIN
    # ==================================================

    def train(
        self,
    ):

        for episode in range(
            1,
            self.episodes + 1,
        ):

            result = self.train_episode(
                episode
            )

            if (
                episode == 1
                or
                episode % 10 == 0
                or
                episode == self.episodes
            ):

                print(
                    f"Episode "
                    f"{episode:04d}/"
                    f"{self.episodes:04d} | "
                    f"Reward="
                    f"{result['reward']:.2f} | "
                    f"Steps="
                    f"{result['steps']} | "
                    f"Completed="
                    f"{result['completed']} | "
                    f"Epsilon="
                    f"{result['epsilon']:.4f}"
                )

        return {
            "rewards":
                self.episode_rewards,

            "steps":
                self.episode_lengths,

            "completions":
                self.episode_completions,
        }

    # ==================================================
    # AGV STATE
    # ==================================================

        # ==================================================
    # MULTI-AGV STATE
    # ==================================================

    def _get_agv_state(
        self,
        state,
        agv_id,
    ):

        state = np.asarray(
            state,
            dtype=np.float32,
        )

        # --------------------------------------------------
        # Own AGV state
        # --------------------------------------------------

        own_start = (
            agv_id * 8
        )

        own_end = (
            own_start + 8
        )

        own_state = state[
            own_start:own_end
        ]

        # --------------------------------------------------
        # Other AGV relative states
        # --------------------------------------------------

        other_features = []

        for other_id in range(
            self.num_agvs
        ):

            if other_id == agv_id:

                continue

            other_start = (
                other_id * 8
            )

            other_end = (
                other_start + 8
            )

            other_state = state[
                other_start:other_end
            ]

            relative_state = (
                other_state
                -
                own_state
            )

            other_features.extend(
                relative_state.tolist()
            )

        # --------------------------------------------------
        # Combined state
        # --------------------------------------------------

        combined_state = np.concatenate(
            [
                own_state,
                np.asarray(
                    other_features,
                    dtype=np.float32,
                ),
            ]
        )

        return tuple(
            np.round(
                combined_state,
                6,
            )
        )

"""
Training loop for the base multi-AGV Q-learning system.
"""

from __future__ import annotations

import numpy as np

from algorithms.q_learning import QLearningAgent
from environment.multi_agv_env import MultiAGVEnv


class QLearningTrainer:

    def __init__(
        self,
        episodes: int = 100,
        max_steps: int = 500,
        num_agvs: int = 3,
        seed: int = 42,
    ):

        self.episodes = int(
            episodes
        )

        self.max_steps = int(
            max_steps
        )

        self.num_agvs = int(
            num_agvs
        )

        self.seed = int(
            seed
        )

        # --------------------------------------------------
        # Environment
        # --------------------------------------------------

        self.env = MultiAGVEnv(
            width=20,
            height=20,
            num_agvs=self.num_agvs,
            max_steps=self.max_steps,
            risk_aware_assignment=True,
        )

        # --------------------------------------------------
        # One Q-learning agent per AGV
        # --------------------------------------------------

        self.agents = [
            QLearningAgent(
                action_count=5,
                learning_rate=0.1,
                discount_factor=0.95,
                epsilon=1.0,
                epsilon_min=0.05,
                epsilon_decay=0.995,
                seed=self.seed + agv_id,
            )
            for agv_id in range(
                self.num_agvs
            )
        ]

        self.episode_rewards = []

        self.episode_lengths = []

        self.episode_completions = []

    # ==================================================
    # TRAIN ONE EPISODE
    # ==================================================

    def train_episode(
        self,
        episode: int,
    ):

        states, _ = self.env.reset(
            seed=self.seed + episode
        )

        total_reward = 0.0

        completed_tasks = 0

        for step in range(
            self.max_steps
        ):

            # --------------------------------------------------
            # Select action independently for every AGV
            # --------------------------------------------------

            actions = np.array(
                [
                    agent.select_action(
                        self._get_agv_state(
                            states,
                            agv_id,
                        ),
                        training=True,
                    )
                    for agv_id, agent
                    in enumerate(
                        self.agents
                    )
                ],
                dtype=np.int64,
            )

            (
                next_states,
                reward,
                terminated,
                truncated,
                info,
            ) = self.env.step(
                actions
            )

            # --------------------------------------------------
            # Update each Q learner
            # --------------------------------------------------

            for agv_id, agent in enumerate(
                self.agents
            ):

                state = self._get_agv_state(
                    states,
                    agv_id,
                )

                next_state = self._get_agv_state(
                    next_states,
                    agv_id,
                )

                agent.update(
                    state=state,
                    action=int(
                        actions[agv_id]
                    ),
                    reward=reward,
                    next_state=next_state,
                    terminated=(
                        terminated
                        or truncated
                    ),
                )

            states = next_states

            total_reward += float(
                reward
            )

            completed_tasks = int(
                info.get(
                    "completed_tasks",
                    0,
                )
            )

            if terminated or truncated:

                break

        # --------------------------------------------------
        # Exploration decay
        # --------------------------------------------------

        for agent in self.agents:

            agent.decay_epsilon()

        self.episode_rewards.append(
            total_reward
        )

        self.episode_lengths.append(
            step + 1
        )

        self.episode_completions.append(
            completed_tasks
        )

        return {
            "reward":
                total_reward,

            "steps":
                step + 1,

            "completed":
                completed_tasks,

            "epsilon":
                self.agents[0].epsilon,
        }

    # ==================================================
    # EVALUATE
    # ==================================================
    def evaluate(
        self,
        episodes=20,
        max_steps=None,
        seed=None,
    ):
        max_steps = int(max_steps or self.max_steps)
        base_seed = self.seed if seed is None else int(seed)
        rewards = []
        lengths = []
        completions = []
        collisions = []

        for episode in range(int(episodes)):
            states, _ = self.env.reset(seed=base_seed + episode)
            total_reward = 0.0
            completed = 0
            collision_count = 0

            for step in range(max_steps):
                actions = np.array([
                    agent.select_action(
                        self._get_agv_state(states, agv_id),
                        training=False,
                    )
                    for agv_id, agent in enumerate(self.agents)
                ], dtype=np.int64)

                next_states, reward, terminated, truncated, info = self.env.step(actions)
                total_reward += float(reward)
                completed = int(info.get('completed_tasks', completed))
                collision_count = int(info.get('episode_collision_count', info.get('collision_count', collision_count)))
                states = next_states

                if terminated or truncated:
                    break

            rewards.append(total_reward)
            lengths.append(step + 1)
            completions.append(completed)
            collisions.append(collision_count)

        return {
            'episodes': int(episodes),
            'max_steps': max_steps,
            'average_reward': float(np.mean(rewards)) if rewards else 0.0,
            'average_steps': float(np.mean(lengths)) if lengths else 0.0,
            'average_completed': float(np.mean(completions)) if completions else 0.0,
            'completion_rate': float(100.0 * np.mean(np.asarray(completions) > 0)) if completions else 0.0,
            'average_collisions': float(np.mean(collisions)) if collisions else 0.0,
            'rewards': rewards,
            'steps': lengths,
            'completions': completions,
            'collisions': collisions,
        }

    def train(

    ):

        for episode in range(

        ):

            if (

                    f"{result['steps']} | "

                )

        return {

                self.episode_lengths,

            "completions":
                self.episode_completions,
        }
